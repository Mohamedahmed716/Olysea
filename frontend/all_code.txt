import { Component, computed, inject } from '@angular/core';
import { toSignal } from '@angular/core/rxjs-interop';
import { NavigationEnd, Router, RouterOutlet } from '@angular/router';
import { filter, map } from 'rxjs';
import { Toast } from './core/services/toast';
import { Footer } from './shared/components/footer/footer';
import { Header } from './shared/components/header/header';

@Component({
  selector: 'app-root',
  imports: [RouterOutlet, Header, Footer],
  templateUrl: './app.html',
  styleUrl: './app.css',
})
export class App {
  private readonly router = inject(Router);
  protected readonly toast = inject(Toast);

  private readonly url = toSignal(
    this.router.events.pipe(
      filter((e): e is NavigationEnd => e instanceof NavigationEnd),
      map((e) => e.urlAfterRedirects),
    ),
    { initialValue: this.router.url },
  );

  /** The public header/footer are hidden on the admin screens, which bring their own chrome. */
  protected readonly showChrome = computed(() => {
    const u = this.url();
    return !(u.startsWith('/admin') || u.startsWith('/dashboard'));
  });
}

import { HttpErrorResponse, HttpInterceptorFn } from '@angular/common/http';
import { inject } from '@angular/core';
import { catchError, throwError } from 'rxjs';
import { Auth } from '../services/auth';

function readCookie(name: string): string | null {
  const match = document.cookie.match(new RegExp('(?:^|; )' + name.replace(/[.$?*|{}()[\]\\/+^]/g, '\\$&') + '=([^;]*)'));
  return match ? decodeURIComponent(match[1]) : null;
}

/**
 * - Sends the session cookie with every /api call.
 * - Copies Spring Security's XSRF-TOKEN cookie into the X-XSRF-TOKEN header on state-changing requests.
 * - Clears the local session when an admin call comes back 401.
 */
export const csrfInterceptor: HttpInterceptorFn = (req, next) => {
  if (!req.url.startsWith('/api')) return next(req);

  const auth = inject(Auth);
  const unsafe = !['GET', 'HEAD', 'OPTIONS'].includes(req.method);
  const token = unsafe ? readCookie('XSRF-TOKEN') : null;

  const cloned = req.clone({
    withCredentials: true,
    setHeaders: token ? { 'X-XSRF-TOKEN': token } : {},
  });

  return next(cloned).pipe(
    catchError((err: unknown) => {
      if (err instanceof HttpErrorResponse && err.status === 401 && req.url.includes('/admin/') && !req.url.endsWith('/admin/login')) {
        auth.expire();
      }
      return throwError(() => err);
    }),
  );
};

import { Injectable, computed, inject, signal } from '@angular/core';
import { Router } from '@angular/router';
import { Observable, catchError, map, of, tap } from 'rxjs';
import { AdminUser, LoginRequest, LoginResponse } from '../models/admin-user.model';
import { Api } from './api';

/** Session state. The session itself is a server-side cookie; this only mirrors "who am I". */
@Injectable({ providedIn: 'root' })
export class Auth {
  private readonly api = inject(Api);
  private readonly router = inject(Router);

  readonly user = signal<AdminUser | null>(null);
  readonly isLoggedIn = computed(() => this.user() !== null);
  private checked = false;

  login(body: LoginRequest): Observable<LoginResponse> {
    return this.api.login(body).pipe(
      tap((res) => {
        if (res?.success) {
          this.user.set(res.user ?? { email: body.email });
          this.checked = true;
        }
      }),
    );
  }

  /** Asks the server whether the session cookie is still valid (cached after the first check). */
  check(force = false): Observable<boolean> {
    if (this.checked && !force) return of(this.isLoggedIn());
    return this.api.me().pipe(
      tap((u) => this.user.set(u)),
      map(() => true),
      catchError(() => {
        this.user.set(null);
        return of(false);
      }),
      tap(() => (this.checked = true)),
    );
  }

  logout(): void {
    const done = () => {
      this.user.set(null);
      this.checked = true;
      this.router.navigateByUrl('/admin/login');
    };
    this.api.logout().subscribe({ next: done, error: done });
  }

  /** Called by the interceptor when the server answers 401 on an admin call. */
  expire(): void {
    this.user.set(null);
    this.checked = true;
  }
}

import { DOCUMENT } from '@angular/common';
import { Injectable, computed, effect, inject, signal } from '@angular/core';
import { Lang, MARQUEE, ProcessStep, STEPS, SWAP, TX } from '../data/translations';

const KEY = 'olysea-lang';

/** English / Arabic (RTL) switching, mirrors the language engine of the original olysea.html. */
@Injectable({ providedIn: 'root' })
export class I18n {
  private readonly doc = inject(DOCUMENT);
  private fontLoaded = false;

  readonly lang = signal<Lang>(this.read());
  readonly isAr = computed(() => this.lang() === 'ar');

  constructor() {
    effect(() => {
      const l = this.lang();
      const root = this.doc.documentElement;
      root.lang = l;
      root.dir = l === 'ar' ? 'rtl' : 'ltr';
      if (l === 'ar' && !this.fontLoaded) {
        this.fontLoaded = true;
        const link = this.doc.createElement('link');
        link.rel = 'stylesheet';
        link.href = 'https://fonts.googleapis.com/css2?family=Cairo:wght@400;500;700;800&display=swap';
        this.doc.head.appendChild(link);
      }
      try {
        localStorage.setItem(KEY, l);
      } catch {
        /* storage can be blocked; the language still works for this visit */
      }
    });
  }

  /** Translate a key; falls back to English, then to the key itself. */
  t(key: string): string {
    return TX[this.lang()][key] ?? TX.en[key] ?? key;
  }

  /** Pick the Arabic value when the site is in Arabic and one exists. */
  pick(en: string | null | undefined, ar?: string | null): string {
    const base = en ?? '';
    return this.isAr() && ar ? ar : base;
  }

  swapWords(): string[] {
    return SWAP[this.lang()];
  }

  marquee(): string[] {
    return MARQUEE[this.lang()];
  }

  steps(): ProcessStep[] {
    return STEPS[this.lang()];
  }

  toggle(): void {
    this.lang.set(this.lang() === 'ar' ? 'en' : 'ar');
  }

  private read(): Lang {
    try {
      return localStorage.getItem(KEY) === 'ar' ? 'ar' : 'en';
    } catch {
      return 'en';
    }
  }
}

export interface ToastItem {
  id: number;
  kind: 'ok' | 'bad';
  title: string;
  message: string;
}

@Injectable({ providedIn: 'root' })
export class Toast {
  readonly items = signal<ToastItem[]>([]);
  private next = 1;

  ok(title: string, message = ''): void {
    this.push('ok', title, message, 4500);
  }

  bad(title: string, message = ''): void {
    this.push('bad', title, message, 7000);
  }

  dismiss(id: number): void {
    this.items.update((l) => l.filter((t) => t.id !== id));
  }

  private push(kind: 'ok' | 'bad', title: string, message: string, ms: number): void {
    const id = this.next++;
    this.items.update((l) => [...l, { id, kind, title, message }]);
    setTimeout(() => this.dismiss(id), ms);
  }
}

/** Reads an image file, scales it down to maxWidth and returns a JPEG data URL. */
export function fileToJpegDataUrl(file: File, maxWidth = 1200, quality = 0.82): Promise<string> {
  return new Promise((resolve, reject) => {
    if (!file.type.startsWith('image/')) {
      reject(new Error('Choose an image file (JPG, PNG or WebP).'));
      return;
    }
    const reader = new FileReader();
    reader.onerror = () => reject(new Error('Could not read that file. Try another image.'));
    reader.onload = () => {
      const img = new Image();
      img.onerror = () => reject(new Error('Could not open that image. Try another file.'));
      img.onload = () => {
        const k = Math.min(1, maxWidth / img.width);
        const canvas = document.createElement('canvas');
        canvas.width = Math.round(img.width * k);
        canvas.height = Math.round(img.height * k);
        const ctx = canvas.getContext('2d');
        if (!ctx) {
          reject(new Error('Could not process that image.'));
          return;
        }
        ctx.fillStyle = '#fff';
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
        resolve(canvas.toDataURL('image/jpeg', quality));
      };
      img.src = reader.result as string;
    };
    reader.readAsDataURL(file);
  });
}

import { Component, DestroyRef, ElementRef, HostListener, computed, inject, signal, viewChild } from '@angular/core';
import { I18n } from '../../core/services/i18n';
import { Reveal } from '../../shared/directives/reveal';

const REDUCED = typeof matchMedia !== 'undefined' && matchMedia('(prefers-reduced-motion: reduce)').matches;

@Component({
  selector: 'app-about',
  imports: [Reveal],
  templateUrl: './about.html',
  styleUrl: './about.css',
})
export class About {
  protected readonly i18n = inject(I18n);
  private readonly destroyRef = inject(DestroyRef);
  private readonly lede = viewChild<ElementRef<HTMLElement>>('lede');

  protected readonly principles = [
    { n: 1, color: '#c4b0ff' },
    { n: 2, color: '#8ecbff' },
    { n: 3, color: '#9d8cff' },
    { n: 4, color: '#3a52ff' },
  ];
  protected readonly current = signal(0);
  private timer?: ReturnType<typeof setInterval>;

  /** The lede is split into words that light up as the reader scrolls. */
  protected readonly words = computed(() => this.i18n.t('a.lede').trim().split(/\s+/));
  private readonly progress = signal(REDUCED ? 1 : 0);
  protected readonly lit = computed(() => Math.round(this.progress() * this.words().length));

  constructor() {
    if (!REDUCED) {
      this.timer = setInterval(() => this.current.update((c) => (c + 1) % 4), 4200);
      this.destroyRef.onDestroy(() => clearInterval(this.timer));
    }
    setTimeout(() => this.onScroll());
  }

  @HostListener('window:scroll')
  @HostListener('window:resize')
  onScroll(): void {
    const el = this.lede()?.nativeElement;
    if (!el || REDUCED) return;
    const r = el.getBoundingClientRect();
    const p = Math.max(0, Math.min(1, (innerHeight * 0.85 - r.top) / (r.height + innerHeight * 0.25)));
    this.progress.set(p);
  }

  protected select(i: number): void {
    clearInterval(this.timer);
    this.current.set(i);
  }
}

import { DatePipe } from '@angular/common';
import { Component, computed, inject, signal } from '@angular/core';
import { CONTACT_STATUSES, ContactRequest, ContactStatus } from '../../../core/models/contact-request.model';
import { Api } from '../../../core/services/api';
import { Toast } from '../../../core/services/toast';
import { Loading } from '../../../shared/components/loading/loading';

@Component({
  selector: 'app-admin-contacts',
  imports: [DatePipe, Loading],
  templateUrl: './contacts.html',
  styleUrl: './contacts.css',
})
export class Contacts {
  private readonly api = inject(Api);
  private readonly toast = inject(Toast);

  protected readonly statuses = CONTACT_STATUSES;
  protected readonly tabs: { value: ContactStatus | 'ALL'; label: string }[] = [
    { value: 'ALL', label: 'All' },
    ...CONTACT_STATUSES.map((s) => ({ value: s, label: s.charAt(0) + s.slice(1).toLowerCase() })),
  ];
  protected readonly loading = signal(true);
  protected readonly failed = signal(false);
  protected readonly items = signal<ContactRequest[]>([]);
  protected readonly filter = signal<ContactStatus | 'ALL'>('ALL');
  protected readonly open = signal<number | null>(null);
  protected readonly pendingDelete = signal<ContactRequest | null>(null);

  protected readonly shown = computed(() => {
    const f = this.filter();
    return [...this.items()]
      .sort((a, b) => (b.createdAt ?? '').localeCompare(a.createdAt ?? ''))
      .filter((c) => f === 'ALL' || (c.status ?? 'NEW') === f);
  });

  constructor() {
    this.api.adminContacts().subscribe({
      next: (list) => { this.items.set(list); this.loading.set(false); },
      error: () => { this.failed.set(true); this.loading.set(false); },
    });
  }

  protected count(s: ContactStatus | 'ALL'): number {
    return s === 'ALL' ? this.items().length : this.items().filter((c) => (c.status ?? 'NEW') === s).length;
  }

  protected toggle(c: ContactRequest): void {
    const isOpen = this.open() === c.id;
    this.open.set(isOpen ? null : (c.id ?? null));
    // opening a new message marks it as read
    if (!isOpen && (c.status ?? 'NEW') === 'NEW') this.setStatus(c, 'READ', true);
  }

  protected setStatus(c: ContactRequest, status: ContactStatus, silent = false): void {
    if (c.id == null) return;
    this.api.setContactStatus(c.id, status).subscribe({
      next: () => {
        this.items.update((l) => l.map((x) => (x.id === c.id ? { ...x, status } : x)));
        if (!silent) this.toast.ok('Status updated', `Marked as ${status.toLowerCase()}.`);
      },
      error: () => this.toast.bad('Couldn’t update status', 'Please try again.'),
    });
  }

  protected reply(c: ContactRequest): string {
    return 'mailto:' + c.email + '?subject=' + encodeURIComponent('Re: your Olysea enquiry (' + c.type + ')');
  }

  protected remove(): void {
    const c = this.pendingDelete();
    this.pendingDelete.set(null);
    if (!c || c.id == null) return;
    this.api.deleteContact(c.id).subscribe({
      next: () => {
        this.items.update((l) => l.filter((x) => x.id !== c.id));
        this.toast.ok('Message deleted');
      },
      error: () => this.toast.bad('Couldn’t delete message', 'Please try again.'),
    });
  }
}

import { HttpErrorResponse } from '@angular/common/http';
import { Component, inject, signal } from '@angular/core';
import { FormBuilder, ReactiveFormsModule } from '@angular/forms';
import { Router, RouterLink } from '@angular/router';
import { Auth } from '../../../core/services/auth';

@Component({
  selector: 'app-login',
  imports: [ReactiveFormsModule, RouterLink],
  templateUrl: './login.html',
  styleUrl: './login.css',
})
export class Login {
  private readonly auth = inject(Auth);
  private readonly router = inject(Router);
  private readonly fb = inject(FormBuilder).nonNullable;

  protected readonly form = this.fb.group({ email: '', password: '', remember: false });
  protected readonly showPassword = signal(false);
  protected readonly busy = signal(false);
  protected readonly message = signal('');
  protected readonly ok = signal(false);

  constructor() {
    // already signed in? go straight to the workspace
    this.auth.check().subscribe((yes) => { if (yes) this.router.navigateByUrl('/admin'); });
  }

  protected submit(): void {
    this.message.set('');
    const { email, password, remember } = this.form.getRawValue();
    if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email.trim())) {
      document.getElementById('email')?.focus();
      this.fail('Enter a valid admin email address.');
      return;
    }
    if (!password) {
      document.getElementById('password')?.focus();
      this.fail('Enter your password to continue.');
      return;
    }

    this.busy.set(true);
    this.auth.login({ email: email.trim(), password, remember }).subscribe({
      next: (res) => {
        this.busy.set(false);
        if (!res?.success) {
          this.fail(res?.message || 'Sign-in is unavailable right now. Please try again later.');
          return;
        }
        this.ok.set(true);
        this.message.set('Signed in successfully. Opening your admin workspace…');
        const target = res.redirect && res.redirect.startsWith('/') && !res.redirect.startsWith('//') ? res.redirect : '/admin';
        this.router.navigateByUrl(target);
      },
      error: (err: HttpErrorResponse) => {
        this.busy.set(false);
        this.form.controls.password.setValue('');
        document.getElementById('password')?.focus();
        if (err.status === 401) this.fail('That email or password doesn’t match. Please try again.');
        else if (err.status === 0) this.fail('Could not connect to the sign-in service. Check that the server is running.');
        else this.fail((err.error && err.error.message) || 'Sign-in is unavailable right now. Please try again later.');
      },
    });
  }

  private fail(text: string): void {
    this.ok.set(false);
    this.message.set(text);
  }
}

@Component({
  selector: 'app-admin-services',
  imports: [ReactiveFormsModule, Loading],
  templateUrl: './services.html',
  styleUrl: './services.css',
})
export class AdminServices {
  private readonly api = inject(Api);
  private readonly toast = inject(Toast);
  private readonly fb = inject(FormBuilder).nonNullable;

  private readonly dialog = viewChild<ElementRef<HTMLDialogElement>>('dialog');
  private readonly delDialog = viewChild<ElementRef<HTMLDialogElement>>('delDialog');

  protected readonly loading = signal(true);
  protected readonly failed = signal(false);
  protected readonly items = signal<Service[]>([]);
  protected readonly editing = signal<Service | null>(null);
  protected readonly toDelete = signal<Service | null>(null);
  protected readonly saving = signal(false);
  protected readonly submitted = signal(false);

  protected readonly form = this.fb.group({
    title: ['', Validators.required],
    description: ['', [Validators.required, Validators.minLength(10)]],
    extra: '',
    titleAr: '',
    descriptionAr: '',
    extraAr: '',
  });

    this.reload();
  }

    this.loading.set(true);
    this.api.adminServices().subscribe({
      next: (l) => { this.items.set(l); this.failed.set(false); this.loading.set(false); },
      error: () => { this.failed.set(true); this.loading.set(false); },
    });

      titleAr: v.titleAr.trim() || null, descriptionAr: v.descriptionAr.trim() || null, extraAr: v.extraAr.trim() || null,
    };

import { Component, computed, inject, signal } from '@angular/core';
import { FormBuilder, ReactiveFormsModule, Validators } from '@angular/forms';
import { CONTACT_TYPES } from '../../core/models/contact-request.model';
import { SITE } from '../../core/data/defaults';
import { Api } from '../../core/services/api';
import { I18n } from '../../core/services/i18n';

@Component({
  selector: 'app-contact',
  imports: [ReactiveFormsModule],
  templateUrl: './contact.html',
  styleUrl: './contact.css',
})
export class Contact {
  protected readonly i18n = inject(I18n);
  private readonly api = inject(Api);
  private readonly fb = inject(FormBuilder).nonNullable;

  protected readonly site = SITE;
  protected readonly types = CONTACT_TYPES;

  protected readonly form = this.fb.group({
    name: ['', [Validators.required, Validators.pattern(/\S/)]],
    email: ['', [Validators.required, Validators.pattern(/^[^@\s]+@[^@\s]+\.[^@\s]+$/)]],
    business: [''],
    type: ['', Validators.required],
    message: ['', [Validators.required, Validators.minLength(10)]],
    website: [''], // honeypot, real visitors never see or fill this
  });

  protected readonly sending = signal(false);
  /** translation key of the status banner, so it re-translates when the language flips */
  protected readonly statusKey = signal<string | null>(null);
  protected readonly statusOk = signal(false);
  private readonly submitted = signal(false);

  protected readonly status = computed(() => (this.statusKey() ? this.i18n.t(this.statusKey()!) : ''));

  /** Error key for a field once it was touched (or after a submit attempt). */
  protected error(field: 'name' | 'email' | 'type' | 'message'): string {
    const c = this.form.controls[field];
    if (!c.invalid || !(c.touched || this.submitted())) return '';
    return 'err.' + (field === 'message' ? 'msg' : field);
  }

  protected submit(): void {
    this.submitted.set(true);
    this.statusKey.set(null);
    if (this.form.invalid) {
      this.form.markAllAsTouched();
      this.show('m.fix', false);
      return;
    }
    const v = this.form.getRawValue();
    if (v.website) return; // bot

    this.sending.set(true);
    this.api
      .sendContact({
        name: v.name.trim(),
        email: v.email.trim(),
        business: v.business.trim() || undefined,
        type: v.type,
        message: v.message.trim(),
        website: v.website,
      })
      .subscribe({
        next: () => {
          this.sending.set(false);
          this.form.reset();
          this.submitted.set(false);
          this.show('m.ok', true);
        },
        error: () => {
          this.sending.set(false);
          this.show('m.err', false);
        },
      });
  }

  private show(key: string, ok: boolean): void {
    this.statusKey.set(key);
    this.statusOk.set(ok);
  }
}

import { Component, DestroyRef, computed, inject, signal } from '@angular/core';
import { RouterLink } from '@angular/router';
import { I18n } from '../../core/services/i18n';
import { Reveal } from '../../shared/directives/reveal';

const REDUCED = typeof matchMedia !== 'undefined' && matchMedia('(prefers-reduced-motion: reduce)').matches;
const STEP_MS = 6000;

@Component({
  selector: 'app-home',
  imports: [RouterLink, Reveal],
  templateUrl: './home.html',
  styleUrl: './home.css',
})
export class Home {
  protected readonly i18n = inject(I18n);
  private readonly destroyRef = inject(DestroyRef);

  // ---- rotating word in the headline ----
  private readonly swapIndex = signal(0);
  protected readonly swapOut = signal(false);
  protected readonly swapPre = signal(false);
  protected readonly swapWord = computed(() => {
    const w = this.i18n.swapWords();
    return w[this.swapIndex() % w.length];
  });

  // ---- hero parallax + fake "project status" card ----
  protected readonly mx = signal(0);
  protected readonly my = signal(0);
  private readonly tick = signal(0);
  protected readonly chips = computed(() => [0, 1, 2].map((i) => Math.max(0, Math.min(3, this.tick() - i * 2))));
  protected readonly progress = computed(() => (this.chips().reduce((a, c) => a + c, 0) / 9) * 100);
  protected readonly rows = ['ui.r1', 'ui.r2', 'ui.r3'];

  // ---- process tabs ----
  protected readonly step = signal(0);
  protected readonly steps = computed(() => this.i18n.steps());
  protected readonly current = computed(() => this.steps()[this.step()]);
  protected readonly stepDur = STEP_MS + 'ms';
  protected readonly stepIdx = [0, 1, 2, 3];
  private paused = false;
  private stepTimer?: ReturnType<typeof setTimeout>;

  protected readonly why = [1, 2, 3, 4];

  constructor() {
    if (!REDUCED) {
      const a = setInterval(() => this.rotateWord(), 2300);
      const b = setInterval(() => this.tick.update((t) => (t + 1) % 10), 1500);
      this.queueStep();
      this.destroyRef.onDestroy(() => {
        clearInterval(a);
        clearInterval(b);
        clearTimeout(this.stepTimer);
      });
    }
  }

  private rotateWord(): void {
    this.swapOut.set(true);
    setTimeout(() => {
      this.swapIndex.update((i) => i + 1);
      this.swapOut.set(false);
      this.swapPre.set(true);
      requestAnimationFrame(() => requestAnimationFrame(() => this.swapPre.set(false)));
    }, 360);
  }

  protected onPointer(e: PointerEvent, el: HTMLElement): void {
    if (e.pointerType !== 'mouse' || REDUCED) return;
    const r = el.getBoundingClientRect();
    this.mx.set(+((e.clientX - r.left) / r.width - 0.5).toFixed(3));
    this.my.set(+((e.clientY - r.top) / r.height - 0.5).toFixed(3));
  }

  protected selectStep(i: number): void {
    this.step.set(i);
    this.queueStep();
  }

  protected pauseSteps(p: boolean): void {
    this.paused = p;
    this.queueStep();
  }

  protected onStepKey(e: KeyboardEvent, i: number): void {
    const rtl = this.i18n.isAr();
    const fwd = rtl ? 'ArrowLeft' : 'ArrowRight';
    const back = rtl ? 'ArrowRight' : 'ArrowLeft';
    let n: number | null = null;
    if (e.key === fwd || e.key === 'ArrowDown') n = (i + 1) % 4;
    else if (e.key === back || e.key === 'ArrowUp') n = (i + 3) % 4;
    else if (e.key === 'Home') n = 0;
    else if (e.key === 'End') n = 3;
    if (n !== null) {
      e.preventDefault();
      this.selectStep(n);
      setTimeout(() => document.getElementById('t' + n)?.focus());
    }
  }

  private queueStep(): void {
    clearTimeout(this.stepTimer);
    if (REDUCED || this.paused) return;
    this.stepTimer = setTimeout(() => this.selectStep((this.step() + 1) % 4), STEP_MS);
  }
}

import { Component, ElementRef, HostListener, computed, effect, inject, input, signal, viewChild } from '@angular/core';
import { FormBuilder, ReactiveFormsModule } from '@angular/forms';
import { DEFAULT_PROJECTS } from '../../core/data/defaults';
import { Project } from '../../core/models/project.model';
import { Api } from '../../core/services/api';
import { I18n } from '../../core/services/i18n';
import { Toast } from '../../core/services/toast';
import { URL_PATTERN, fileToJpegDataUrl } from '../../core/utils/image';
import { Loading } from '../../shared/components/loading/loading';
import { ProjectPreview } from '../../shared/components/project-preview/project-preview';

const REDUCED = typeof matchMedia !== 'undefined' && matchMedia('(prefers-reduced-motion: reduce)').matches;

/**
 * The "Demos" carousel. With `[editable]="true"` (used by the signed-in live editor) every card gets a delete
 * button and a final "Add new project" card opens a dialog — the behaviour of the original olysea-admin.html.
 */
@Component({
  selector: 'app-projects',
  imports: [ReactiveFormsModule, Loading, ProjectPreview],
  templateUrl: './projects.html',
  styleUrl: './projects.css',
})
export class Projects {
  readonly editable = input(false);

  protected readonly i18n = inject(I18n);
  private readonly api = inject(Api);
  private readonly toast = inject(Toast);
  private readonly fb = inject(FormBuilder).nonNullable;

  private readonly track = viewChild<ElementRef<HTMLElement>>('track');
  private readonly addDialog = viewChild<ElementRef<HTMLDialogElement>>('addDialog');
  private readonly delDialog = viewChild<ElementRef<HTMLDialogElement>>('delDialog');

  protected readonly colors = ['#ff9ecb', '#8ecbff', '#c4b0ff', '#9aa9ff', '#ffd1e8', '#b9e3ff'];
  protected readonly items = signal<Project[]>([]);
  protected readonly loading = signal(true);
  protected readonly idx = signal(0);
  protected readonly total = computed(() => this.items().length + (this.editable() ? 1 : 0));
  protected readonly counter = computed(() => {
    const pad = (n: number) => String(n).padStart(2, '0');
    const n = this.items().length;
    return this.idx() < n ? `${pad(this.idx() + 1)} / ${pad(n)}` : `NEW / ${pad(n)}`;
  });
  protected readonly progress = computed(() => ((this.idx() + 1) / Math.max(1, this.total())) * 100);

  // ---- add / delete state ----
  protected readonly form = this.fb.group({ category: '', name: '', description: '', url: '', categoryAr: '', descriptionAr: '' });
  protected readonly errors = signal<Record<string, string>>({});
  protected readonly imageData = signal<string | null>(null);
  protected readonly saving = signal(false);
  protected readonly toDelete = signal<Project | null>(null);

  constructor() {
    this.load();
    // after every data / language / mode change, re-centre the carousel on the active card
    effect(() => {
      this.items();
      this.i18n.lang();
      this.editable();
      setTimeout(() => this.init());
    });
  }

  private load(): void {
    this.api.getProjects().subscribe({
      next: (list) => {
        this.items.set(list);
        this.loading.set(false);
      },
      error: () => {
        if (this.editable()) this.toast.bad('Could not load projects', 'Check that the server is running.');
        else this.items.set(DEFAULT_PROJECTS);
        this.loading.set(false);
      },
    });
  }

  // ---------------- carousel mechanics (direction agnostic: works in LTR and RTL) ----------------
  private cards(): HTMLElement[] {
    return Array.from(this.track()?.nativeElement.querySelectorAll<HTMLElement>('.card') ?? []);
  }

  private init(): void {
    const cards = this.cards();
    if (!cards.length) return;
    this.centerOn(cards[Math.min(this.idx(), cards.length - 1)], true);
    this.update();
  }

  private centerOn(el: HTMLElement, instant = false): void {
    const tr = this.track()?.nativeElement;
    if (!tr) return;
    const r = el.getBoundingClientRect();
    const t = tr.getBoundingClientRect();
    const d = r.left + r.width / 2 - (t.left + t.width / 2);
    if (instant) {
      tr.style.scrollBehavior = 'auto';
      tr.scrollLeft += d;
      tr.style.scrollBehavior = '';
    } else {
      tr.scrollBy({ left: d, behavior: REDUCED ? 'auto' : 'smooth' });
    }
  }

  protected update(): void {
    const tr = this.track()?.nativeElement;
    const cards = this.cards();
    if (!tr || !cards.length) return;
    const t = tr.getBoundingClientRect();
    const c = t.left + t.width / 2;
    let best = 0;
    let bd = Infinity;
    cards.forEach((el, i) => {
      const r = el.getBoundingClientRect();
      const d = Math.abs(r.left + r.width / 2 - c);
      if (d < bd) { bd = d; best = i; }
    });
    this.idx.set(best);
  }

  protected go(i: number): void {
    const cards = this.cards();
    if (!cards.length) return;
    this.centerOn(cards[Math.max(0, Math.min(cards.length - 1, i))]);
  }

  protected onScroll(): void {
    requestAnimationFrame(() => this.update());
  }

  @HostListener('window:resize')
  onResize(): void {
    this.update();
  }

  // mouse drag (touch uses native swipe)
  private down = false;
  private startX = 0;
  private startLeft = 0;
  protected moved = false;
  protected dragging = signal(false);

  protected onDown(e: PointerEvent): void {
    if (e.pointerType !== 'mouse') return;
    this.down = true;
    this.moved = false;
    this.startX = e.clientX;
    this.startLeft = this.track()?.nativeElement.scrollLeft ?? 0;
  }

  @HostListener('window:pointermove', ['$event'])
  onMove(e: PointerEvent): void {
    if (!this.down) return;
    const dx = e.clientX - this.startX;
    if (Math.abs(dx) > 5) { this.moved = true; this.dragging.set(true); }
    const tr = this.track()?.nativeElement;
    if (tr) tr.scrollLeft = this.startLeft - dx;
  }

  @HostListener('window:pointerup')
  @HostListener('window:pointercancel')
  onUp(): void {
    if (!this.down) return;
    this.down = false;
    this.dragging.set(false);
    if (this.moved) this.go(this.idx());
  }

  protected onTrackClick(e: Event): void {
    if (this.moved) { e.preventDefault(); this.moved = false; }
  }

  // ---------------- editing (admin) ----------------
  protected openAdd(): void {
    this.form.reset();
    this.errors.set({});
    this.imageData.set(null);
    this.addDialog()?.nativeElement.showModal();
  }

  protected closeAdd(): void {
    this.addDialog()?.nativeElement.close();
  }

  protected async onFile(e: Event): Promise<void> {
    const file = (e.target as HTMLInputElement).files?.[0];
    this.imageData.set(null);
    this.setError('image', '');
    if (!file) return;
    try {
      this.imageData.set(await fileToJpegDataUrl(file));
    } catch (err) {
      this.setError('image', err instanceof Error ? err.message : 'Could not use that image.');
    }
  }

  private setError(key: string, msg: string): void {
    this.errors.update((e) => ({ ...e, [key]: msg }));
  }

  protected submitAdd(): void {
    if (this.saving()) return;
    const v = this.form.getRawValue();
    const errs: Record<string, string> = {};
    if (!v.category.trim()) errs['category'] = 'Enter a category.';
    if (!v.name.trim()) errs['name'] = 'Enter a project name.';
    if (v.description.trim().length < 10) errs['description'] = 'Write a short description (at least 10 characters).';
    if (!URL_PATTERN.test(v.url.trim())) errs['url'] = 'Enter a full link starting with https://';
    if (!this.imageData()) errs['image'] = 'Add a photo of the demo.';
    this.errors.set(errs);
    if (Object.keys(errs).length) return;

    this.saving.set(true);
    this.api
      .createProject({
        category: v.category.trim(),
        name: v.name.trim(),
        description: v.description.trim(),
        url: v.url.trim(),
        image: this.imageData(),
        previewKey: null,
        categoryAr: v.categoryAr.trim() || null,
        descriptionAr: v.descriptionAr.trim() || null,
      })
      .subscribe({
        next: (created) => {
          this.saving.set(false);
          this.closeAdd();
          this.items.update((l) => [...l, created]);
          this.idx.set(this.items().length - 1);
          this.toast.ok('Project added', `“${created.name}” is now on the website.`);
        },
        error: () => {
          this.saving.set(false);
          this.toast.bad('Couldn’t add project', 'Something went wrong, so nothing was saved. Please try again.');
        },
      });
  }

  protected askDelete(p: Project): void {
    this.toDelete.set(p);
    this.delDialog()?.nativeElement.showModal();
  }

  protected cancelDelete(): void {
    this.toDelete.set(null);
    this.delDialog()?.nativeElement.close();
  }

  protected confirmDelete(): void {
    const p = this.toDelete();
    this.delDialog()?.nativeElement.close();
    this.toDelete.set(null);
    if (!p || p.id == null) return;
    this.api.deleteProject(p.id).subscribe({
      next: () => {
        this.items.update((l) => l.filter((x) => x.id !== p.id));
        this.idx.update((i) => Math.min(i, this.items().length));
        this.toast.ok('Project deleted', `“${p.name}” was removed from the website.`);
      },
      error: () => this.toast.bad('Couldn’t delete project', `Something went wrong, so “${p.name}” was not removed. Please try again.`),
    });
  }
}

import { Component, ElementRef, HostListener, inject, signal } from '@angular/core';
import { RouterLink, RouterLinkActive } from '@angular/router';
import { I18n } from '../../../core/services/i18n';

@Component({
  selector: 'app-header',
  imports: [RouterLink, RouterLinkActive],
  templateUrl: './header.html',
  styleUrl: './header.css',
})
export class Header {
  protected readonly i18n = inject(I18n);
  private readonly host = inject<ElementRef<HTMLElement>>(ElementRef);

  protected readonly open = signal(false);
  protected readonly small = signal(false);

  @HostListener('window:scroll')
  onScroll(): void {
    this.small.set(window.scrollY > 60);
  }

  @HostListener('document:click', ['$event'])
  onDocClick(e: MouseEvent): void {
    if (this.open() && !this.host.nativeElement.contains(e.target as Node)) this.open.set(false);
  }

  @HostListener('document:keydown.escape')
  onEscape(): void {
    this.open.set(false);
  }

  protected toggle(): void {
    this.open.update((v) => !v);
  }

  protected close(): void {
    this.open.set(false);
  }
}

import { Component, ViewEncapsulation, computed, inject, input } from '@angular/core';
import { DomSanitizer, SafeHtml } from '@angular/platform-browser';
import { PREVIEWS } from '../../../core/data/previews';
import { Project } from '../../../core/models/project.model';

/**
 * Browser-window mock showing a demo site. Uses the uploaded screenshot when there is one, otherwise a built-in
 * illustrated preview picked by `previewKey`.
 *
 * SECURITY: the illustrated previews are static HTML written in core/data/previews.ts and are the ONLY strings
 * trusted as HTML here. Anything that comes from the API (name, url, image) is bound as text / src and never as HTML.
 */
@Component({
  selector: 'app-project-preview',
  templateUrl: './project-preview.html',
  styleUrl: './project-preview.css',
  encapsulation: ViewEncapsulation.None,
})
export class ProjectPreview {
  private readonly sanitizer = inject(DomSanitizer);

  readonly project = input.required<Project>();

  protected readonly host = computed(() => this.project().url.replace(/^https?:\/\//, ''));

  protected readonly image = computed(() => {
    const img = this.project().image;
    // only allow image data URLs or http(s) URLs
    return img && /^(data:image\/|https?:\/\/)/i.test(img) ? img : null;
  });

  protected readonly illustration = computed<SafeHtml | null>(() => {
    const key = this.project().previewKey;
    const html = key ? PREVIEWS[key] : undefined;
    return html ? this.sanitizer.bypassSecurityTrustHtml(html) : null;
  });
}

==> src/app/shared/directives/reveal.ts <==
import { Directive, ElementRef, OnDestroy, OnInit, inject } from '@angular/core';

/** Fades/slides an element in the first time it scrolls into view (the `.rv` effect of the original site). */
@Directive({ selector: '[appReveal]', host: { class: 'rv' } })
export class Reveal implements OnInit, OnDestroy {
  private readonly el = inject<ElementRef<HTMLElement>>(ElementRef);
  private io?: IntersectionObserver;

  ngOnInit(): void {
    const node = this.el.nativeElement;
    if (typeof IntersectionObserver === 'undefined') {
      node.classList.add('in');
      return;
    }
    this.io = new IntersectionObserver(
      (entries) => {
        for (const e of entries) {
          if (e.isIntersecting) {
            node.classList.add('in');
            this.io?.disconnect();
          }
        }
      },
      { threshold: 0.15 },
    );
    this.io.observe(node);
  }

  ngOnDestroy(): void {
    this.io?.disconnect();
  }
}
